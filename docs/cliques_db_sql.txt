CREATE DATABASE cliques_db
	CHARACTER SET utf8mb4
	COLLATE utf8mb4_unicode_ci;

USE cliques_db;

CREATE TABLE users ( 
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(20) NOT NULL UNIQUE,
    userEmail VARCHAR(50) NOT NULL UNIQUE,
    creationDate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    userBirthday DATE NULL,
    -- passwordHash VARCHAR(20) NOT NULL, + salt VARCHAR(40) NOT NULL,
    warningStatus BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE groups (
	id INT AUTO_INCREMENT PRIMARY KEY,
	name VARCHAR(50) NOT NULL,
    creationDate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    adultGate BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE channels (
	id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL,
    groupId INT NOT NULL,
    adultGate BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE memberships (
	userId INT NOT NULL,
    groupId INT NOT NULL,
    channelId INT NOT NULL,
    modStatus BOOLEAN NOT NULL DEFAULT FALSE,
    joinDate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    leaveDate DATETIME NULL
);

CREATE TABLE messages (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    body TEXT NOT NULL,
    sentTime DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    senderId INT NOT NULL,
    channelId INT NULL,
    edited BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE TABLE attachments (
    id INT AUTO_INCREMENT PRIMARY KEY,
    messageId BIGINT NOT NULL,
    format VARCHAR(20) NOT NULL,
    -- body TEXT NOT NULL,
    sizeInKilobytes BIGINT NOT NULL
);
